import os
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
STORAGE_DIR = Path(os.getenv("CLIPFORGE_STORAGE_DIR", ROOT / "storage"))

SUBDIRS = [
    "original_videos", "generated_clips", "edited_clips", "thumbnails", "subtitles",
    "audio", "projects", "collections", "exports", "temporary",
]

MEDIA_TYPES = {
    ".mp4": "video/mp4",
    ".mov": "video/quicktime",
    ".mkv": "video/x-matroska",
    ".avi": "video/x-msvideo",
    ".webm": "video/webm",
    ".mp3": "audio/mpeg",
    ".wav": "audio/wav",
}
ALLOWED_EXTENSIONS = set(MEDIA_TYPES)


def max_upload_bytes() -> int:
    return int(float(os.getenv("MAX_UPLOAD_GB", "5")) * 1024**3)


def ensure_storage_dirs() -> None:
    for d in SUBDIRS:
        (STORAGE_DIR / d).mkdir(parents=True, exist_ok=True)


def safe_display_name(name: str) -> str:
    """Only used for DISPLAY. Files on disk always get random UUID names."""
    name = os.path.basename(name.replace("\\", "/"))
    name = re.sub(r'[\x00-\x1f<>:"|?*]', "_", name).strip(" .")
    return name[:200] or "video"

def folder_size(path: Path) -> int:
    total = 0
    for dirpath, _dirs, files in os.walk(path):
        for f in files:
            try:
                total += os.path.getsize(os.path.join(dirpath, f))
            except OSError:
                pass
    return total

DEFAULT_MODEL = "gemini-3.1-flash-lite"


def get_model() -> str:
    return os.getenv("GEMINI_MODEL", DEFAULT_MODEL)


def is_configured() -> bool:
    key = os.getenv("GEMINI_API_KEY", "")
    return bool(key) and key != "your_key_here"


def ask(prompt: str) -> str:
    """Send TEXT ONLY to Gemini (transcripts, never video files)."""
    if not is_configured():
        raise RuntimeError("GEMINI_API_KEY is not set in .env")
    from google import genai

    client = genai.Client(api_key=os.environ["GEMINI_API_KEY"])
    resp = client.models.generate_content(model=get_model(), contents=prompt)
    return resp.text or ""

def _mmss(sec: float) -> str:
    s = int(sec)
    return f"{s // 60}:{s % 60:02d}"

def _file_or_404(rel, what: str) -> Path:
    if not rel:
        raise HTTPException(404, f"{what} is not ready")
    try:
        p = resolve_storage(rel)
    except ValueError:
        raise HTTPException(400, "Invalid stored path")
    if not p.is_file():
        raise HTTPException(404, f"{what} is missing on disk")
    return p


@router.post("")
def create_clip(body: ClipCreate, db: Session = Depends(get_db)):
    v = db.get(Video, body.video_id)
    if not v:
        raise HTTPException(404, "Video not found")
    if v.width is None:
        raise HTTPException(400, "Clips can only be cut from video files (this file is audio only)")
    start, end = body.start_sec, body.end_sec
    if v.duration_sec is not None:
        if start >= v.duration_sec:
            raise HTTPException(400, "Start is beyond the end of the video")
        end = min(end, v.duration_sec)
    if end - start < MIN_CLIP_SEC:
        raise HTTPException(400, f"Clip must be at least {MIN_CLIP_SEC} seconds long")

    title = body.title.strip() or f"{Path(v.original_name).stem} ({_mmss(start)}-{_mmss(end)})"
    clip = Clip(video_id=v.id, title=title[:200], start_sec=round(start, 3), end_sec=round(end, 3),
                duration_sec=round(end - start, 3), status="pending")
    db.add(clip)
    db.flush()
    job = Job(type="clip_export", video_id=v.id, status="queued",
              payload=json.dumps({"clip_id": clip.id, "mode": body.mode}))
    db.add(job)
    db.commit()
    db.refresh(clip)
    db.refresh(job)
    return {"clip": clip_dict(clip), "job_id": job.id}

@router.post("/{job_id}/cancel")
def cancel_job(job_id: int, db: Session = Depends(get_db)):
    j = db.get(Job, job_id)
    if not j:
        raise HTTPException(404, "Job not found")
    if j.status in TERMINAL:
        raise HTTPException(409, "Job already finished")
    try:
        clip_id = json.loads(j.payload or "{}").get("clip_id")
    except ValueError:
        clip_id = None
    db.execute(
        update(Job).where(Job.id == job_id, Job.status.in_(["queued", "processing"]))
        .values(status="cancelled", error="Cancelled by user", updated_at=utcnow())
    )
    if clip_id:
        db.execute(update(Clip).where(Clip.id == clip_id, Clip.status.in_(["pending", "processing"]))
                   .values(status="cancelled"))
    db.commit()
    db.refresh(j)
    return job_dict(j)


@router.get("/{job_id}/events")
async def job_events(job_id: int, request: Request):
    """Server-Sent Events: streams job progress until the job finishes."""
    with SessionLocal() as db:
        if db.get(Job, job_id) is None:
            raise HTTPException(404, "Job not found")

    async def gen():
        last = None
        last_send = time.monotonic()
        while True:
            if await request.is_disconnected():
                break
            with SessionLocal() as db:
                j = db.get(Job, job_id)
                data = job_dict(j) if j else None
            if data is None:
                yield 'data: {"status": "failed", "error": "Job disappeared"}\n\n'
                break
            payload = json.dumps(data)
            if payload != last:
                yield f"data: {payload}\n\n"
                last, last_send = payload, time.monotonic()
            elif time.monotonic() - last_send > 15:
                yield ": ping\n\n"
                last_send = time.monotonic()
            if data["status"] in TERMINAL:
                break
            await asyncio.sleep(0.5)

    return StreamingResponse(
        gen(), media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},
    )

def _proxy_file(v: Video):
    if not v.proxy_path:
        return None
    try:
        p = resolve_storage(v.proxy_path)
    except ValueError:
        return None
    return p if p.is_file() else None


@router.post("/{video_id}/proxy")
def request_proxy(video_id: int, db: Session = Depends(get_db)):
    """Make a browser-friendly preview copy (used when the original cannot play in the browser)."""
    v = db.get(Video, video_id)
    if not v:
        raise HTTPException(404, "Video not found")
    if v.width is None:
        raise HTTPException(400, "Audio files do not need a video preview")
    if _proxy_file(v):
        return {"ready": True, "job_id": None}
    active = db.scalars(
        select(Job).where(Job.type == "proxy", Job.video_id == video_id, Job.status.in_(["queued", "processing"]))
    ).first()
    if active:
        return {"ready": False, "job_id": active.id}
    job = Job(type="proxy", video_id=video_id, status="queued", payload=json.dumps({"video_id": video_id}))
    db.add(job)
    db.commit()
    db.refresh(job)
    return {"ready": False, "job_id": job.id}


@router.get("/{video_id}/proxy")
def get_proxy(video_id: int, db: Session = Depends(get_db)):
    v = db.get(Video, video_id)
    if not v:
        raise HTTPException(404, "Video not found")
    p = _proxy_file(v)
    if not p:
        raise HTTPException(404, "Preview is not ready")
    return FileResponse(p, media_type="video/mp4")

import shutil
import uuid
from pathlib import Path

from fastapi import APIRouter, Depends, File, HTTPException, UploadFile

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from backend import config
from backend.database.db import get_db
from backend.database.models import Video
from backend.video_processing.ffprobe import ProbeError, probe
from backend.services.cleanup import cleanup_video_files

router = APIRouter(prefix="/api/videos", tags=["videos"])
CHUNK = 1024 * 1024


class UploadRejected(Exception):
    def __init__(self, code: str, message: str):
        self.code = code
        self.message = message


def video_dict(v: Video) -> dict:
    return {
        "id": v.id,
        "name": v.original_name,
        "size_bytes": v.size_bytes,
        "duration_sec": v.duration_sec,
        "width": v.width,
        "height": v.height,
        "format": v.format,
        "status": v.status,
        "has_video": v.width is not None,
        "created_at": v.created_at.isoformat(),
    }


def _save_one(upload: UploadFile, db: Session) -> dict:
    display = config.safe_display_name(upload.filename or "")
    ext = Path(display).suffix.lower()
    if ext not in config.ALLOWED_EXTENSIONS:
        allowed = ", ".join(sorted(config.ALLOWED_EXTENSIONS))
        raise UploadRejected("bad_type", f"Unsupported file type '{ext or 'none'}'. Allowed: {allowed}")

    tmp = config.STORAGE_DIR / "temporary" / f"{uuid.uuid4().hex}.part"
    limit = config.max_upload_bytes()
    size = 0
    try:
        with open(tmp, "wb") as out:
            while True:
                chunk = upload.file.read(CHUNK)
                if not chunk:
                    break
                size += len(chunk)
                if size > limit:
                    raise UploadRejected("too_large", "File is larger than the allowed upload limit (MAX_UPLOAD_GB in .env)")
                out.write(chunk)
        if size == 0:
            raise UploadRejected("empty", "File is empty")
        try:
            info = probe(tmp)
        except ProbeError as e:
            raise UploadRejected("invalid_media", str(e))

        final_name = f"{uuid.uuid4().hex}{ext}"
        final = config.STORAGE_DIR / "original_videos" / final_name
        shutil.move(str(tmp), str(final))
    finally:
        tmp.unlink(missing_ok=True)

    try:
        v = Video(
            original_name=display,
            stored_path=f"original_videos/{final_name}",
            size_bytes=size,
            duration_sec=info["duration_sec"],
            width=info["width"],
            height=info["height"],
            format=ext.lstrip("."),
            status="uploaded",
        )
        db.add(v)
        db.commit()
        db.refresh(v)
    except Exception:
        db.rollback()
        final.unlink(missing_ok=True)
        raise
    return video_dict(v)


@router.post("/upload")
def upload_videos(files: list[UploadFile] = File(...), db: Session = Depends(get_db)):
    uploaded, errors = [], []
    for f in files:
        try:
            uploaded.append(_save_one(f, db))
        except UploadRejected as e:
            errors.append({"filename": f.filename, "code": e.code, "error": e.message})
    if uploaded or not errors:
        status = 200
    elif any(e["code"] == "too_large" for e in errors):
        status = 413
    else:
        status = 400
    return JSONResponse({"uploaded": uploaded, "errors": errors}, status_code=status)

import os
from pathlib import Path
from sqlalchemy import create_engine, event, inspect, text
from sqlalchemy.orm import sessionmaker, DeclarativeBase

ROOT = Path(__file__).resolve().parents[2]
DB_PATH = Path(os.getenv("CLIPFORGE_DB_PATH", ROOT / "storage" / "clipforge.db"))
DB_PATH.parent.mkdir(parents=True, exist_ok=True)

engine = create_engine(f"sqlite:///{DB_PATH}", connect_args={"check_same_thread": False})


@event.listens_for(engine, "connect")
def _set_pragmas(dbapi_conn, _record):
    cur = dbapi_conn.cursor()
    cur.execute("PRAGMA journal_mode=WAL")
    cur.execute("PRAGMA foreign_keys=ON")
    cur.close()


SessionLocal = sessionmaker(bind=engine, autoflush=False, expire_on_commit=False)


class Base(DeclarativeBase):
    pass


def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


# create_all() never alters existing tables, so new columns are added here.
_NEW_COLUMNS = {
    "videos": {"proxy_path": "VARCHAR(1000)"},
    "clips": {"thumbnail_path": "VARCHAR(1000)"},
    "jobs": {"payload": "TEXT"},
}


def _ensure_columns():
    insp = inspect(engine)
    with engine.begin() as conn:
        for table, cols in _NEW_COLUMNS.items():
            have = {c["name"] for c in insp.get_columns(table)}

"""FFmpeg helpers: cut clips, build browser-preview proxies, grab frames."""
import collections
import re
import shutil
import subprocess
from pathlib import Path
from typing import Callable, Optional

ProgressCb = Optional[Callable[[float], None]]  # receives 0.0 .. 1.0

_NO_WINDOW = getattr(subprocess, "CREATE_NO_WINDOW", 0)
_PROGRESS_KEYS = re.compile(
    r"^(frame|fps|stream_\d+_\d+_q|bitrate|total_size|out_time_us|out_time_ms|out_time|"
    r"dup_frames|drop_frames|speed|progress)="
)


class FfmpegError(Exception):
    pass


def _exe() -> str:
    exe = shutil.which("ffmpeg")
    if not exe:
        raise FfmpegError("ffmpeg not found. Install FFmpeg and restart VS Code.")
    return exe


def _run(args: list, total_sec: float, on_progress: ProgressCb = None) -> None:
    cmd = [_exe(), "-hide_banner", "-nostdin", "-y", "-progress", "pipe:1", "-nostats", *args]
    tail = collections.deque(maxlen=25)
    proc = subprocess.Popen(
        cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
        text=True, encoding="utf-8", errors="replace", creationflags=_NO_WINDOW,
    )
    try:
        assert proc.stdout is not None
        for raw in proc.stdout:
            line = raw.strip()
            if not line:
                continue
            if line.startswith(("out_time_us=", "out_time_ms=")):
                if on_progress and total_sec > 0:
                    try:
                        us = int(line.split("=", 1)[1])
                    except ValueError:
                        continue
                    on_progress(min(max(us / 1_000_000 / total_sec, 0.0), 1.0))
            elif not _PROGRESS_KEYS.match(line):
                tail.append(line)
        code = proc.wait()
    finally:
        if proc.poll() is None:  # e.g. cancelled: stop ffmpeg
            proc.kill()
            proc.wait()
    if code != 0:
        raise FfmpegError("\n".join(tail)[-800:] or f"ffmpeg exited with code {code}")


def cut_clip(src, dst, start: float, end: float, mode: str = "accurate", on_progress: ProgressCb = None) -> None:
    """Cut [start, end] seconds of src into an MP4 at dst. Original is never modified.
    mode 'accurate' = re-encode (frame exact). mode 'fast' = stream copy (instant, keyframe aligned)."""
    dur = float(end) - float(start)
    if dur <= 0:
        raise ValueError("end must be greater than start")
    args = ["-ss", f"{start:.3f}", "-i", str(src), "-t", f"{dur:.3f}", "-map", "0:v:0", "-map", "0:a:0?"]
    if mode == "fast":
        args += ["-c", "copy", "-avoid_negative_ts", "make_zero"]
    else:
        args += [
            "-vf", "scale=trunc(iw/2)*2:trunc(ih/2)*2",
            "-c:v", "libx264", "-preset", "veryfast", "-crf", "20", "-pix_fmt", "yuv420p",
            "-c:a", "aac", "-b:a", "160k",
        ]
    args += ["-movflags", "+faststart", str(dst)]
    _run(args, dur, on_progress)


def make_proxy(src, dst, total_sec: float, on_progress: ProgressCb = None) -> None:
    """Small H.264/AAC MP4 (max 720p) so any browser can preview MOV/MKV/AVI/HEVC files."""
    args = [
        "-i", str(src), "-map", "0:v:0", "-map", "0:a:0?",
        "-vf", "scale=-2:'min(720,trunc(ih/2)*2)'",
        "-c:v", "libx264", "-preset", "veryfast", "-crf", "28", "-pix_fmt", "yuv420p",
        "-c:a", "aac", "-b:a", "128k", "-movflags", "+faststart", str(dst),
    ]
    _run(args, total_sec, on_progress)


def grab_frame(src, dst, at_sec: float = 0.0) -> None:
    dst = Path(dst)
    dst.unlink(missing_ok=True)
    cmd = [
        _exe(), "-hide_banner", "-nostdin", "-y", "-ss", f"{max(at_sec, 0.0):.3f}", "-i", str(src),
        "-frames:v", "1", "-vf", "scale=480:-2", "-q:v", "4", str(dst),
    ]
    try:
        r = subprocess.run(cmd, capture_output=True, text=True, encoding="utf-8", errors="replace",
                           timeout=60, creationflags=_NO_WINDOW)
    except subprocess.TimeoutExpired:
        raise FfmpegError("Frame grab timed out")
    if r.returncode != 0 or not dst.is_file() or dst.stat().st_size == 0:
        dst.unlink(missing_ok=True)
        raise FfmpegError((r.stderr or "Could not grab a frame")[-400:])

import json
import shutil
import subprocess


class ProbeError(Exception):
    pass


def probe(path) -> dict:
    exe = shutil.which("ffprobe")
    if not exe:
        raise ProbeError("ffprobe not found. Install FFmpeg and restart VS Code.")
    try:
        r = subprocess.run(
            [exe, "-v", "error", "-print_format", "json", "-show_format", "-show_streams", str(path)],
            capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=60,
        )
    except subprocess.TimeoutExpired:
        raise ProbeError("ffprobe timed out while reading the file")
    if r.returncode != 0:
        raise ProbeError("This is not a valid audio/video file")
    try:
        data = json.loads(r.stdout or "{}")
    except json.JSONDecodeError:
        raise ProbeError("Could not read media information")

    streams = data.get("streams", [])
    video = next(
        (s for s in streams
         if s.get("codec_type") == "video" and s.get("disposition", {}).get("attached_pic", 0) != 1),
        None,
    )
    audio = next((s for s in streams if s.get("codec_type") == "audio"), None)
    if not video and not audio:
        raise ProbeError("No audio or video stream found in this file")

    raw = data.get("format", {}).get("duration") or (video or audio).get("duration")
    try:
        duration = float(raw) if raw is not None else None
    except (TypeError, ValueError):
        duration = None

    return {
        "duration_sec": duration,
        "width": video.get("width") if video else None,
        "height": video.get("height") if video else None,
    }

def _aspect(w, h):
    if not w or not h:
        return None
    r = w / h
    for name, val in (("16:9", 16 / 9), ("9:16", 9 / 16), ("1:1", 1.0), ("4:3", 4 / 3), ("3:4", 3 / 4), ("21:9", 21 / 9)):
        if abs(r - val) < 0.03:
            return name
    return "custom"


def _set_clip(clip_id: int, **values) -> bool:
    with SessionLocal() as db:
        clip = db.get(Clip, clip_id)
        if clip is None:
            return False
        for k, v in values.items():
            setattr(clip, k, v)
        db.commit()
        return True


def _unlink(*paths) -> None:
    for p in paths:
        if p is not None:
            try:
                p.unlink(missing_ok=True)
            except OSError:
                pass


@register("clip_export")
def clip_export(ctx: JobContext) -> None:
    clip_id = int(ctx.payload["clip_id"])
    mode = ctx.payload.get("mode", "accurate")
    with SessionLocal() as db:
        clip = db.get(Clip, clip_id)
        if clip is None:
            raise RuntimeError("Clip record not found")
        video = db.get(Video, clip.video_id)
        if video is None:
            raise RuntimeError("Source video not found")
        stored = video.stored_path
        start, end = clip.start_sec, clip.end_sec
        clip.status = "processing"
        db.commit()

    src = resolve_storage(stored)
    if not src.is_file():
        _set_clip(clip_id, status="failed")
        raise RuntimeError("Source video file is missing on disk")

    name = uuid.uuid4().hex
    dst = config.STORAGE_DIR / "generated_clips" / f"{name}.mp4"
    thumb = config.STORAGE_DIR / "thumbnails" / f"{name}.jpg"
    try:
        if mode == "fast":
            try:
                cut_clip(src, dst, start, end, "fast", ctx.progress)
            except FfmpegError:  # codec not allowed in MP4 container: fall back to re-encode
                _unlink(dst)
                cut_clip(src, dst, start, end, "accurate", ctx.progress)
        else:
            cut_clip(src, dst, start, end, "accurate", ctx.progress)

        info = probe(dst)
        thumb_ok = True
        try:
            grab_frame(dst, thumb, min(1.0, (end - start) / 2))
        except FfmpegError:
            thumb_ok = False

        ok = _set_clip(
            clip_id,
            status="ready",
            file_path=f"generated_clips/{name}.mp4",
            thumbnail_path=f"thumbnails/{name}.jpg" if thumb_ok else None,
            width=info["width"],
            height=info["height"],
            aspect_ratio=_aspect(info["width"], info["height"]),
            duration_sec=info["duration_sec"] or (end - start),
        )
        if not ok:
            raise RuntimeError("Clip was deleted while exporting")
    except JobCancelled:
        _unlink(dst, thumb)
        _set_clip(clip_id, status="cancelled")
        raise
    except Exception:
        _unlink(dst, thumb)
        _set_clip(clip_id, status="failed")
        raise


@register("proxy")
def build_proxy(ctx: JobContext) -> None:
    video_id = int(ctx.payload["video_id"])
    with SessionLocal() as db:
        v = db.get(Video, video_id)
        if v is None:
            raise RuntimeError("Video not found")
        stored, duration = v.stored_path, (v.duration_sec or 0.0)
    src = resolve_storage(stored)
    if not src.is_file():
        raise RuntimeError("Source video file is missing on disk")

    (config.STORAGE_DIR / "previews").mkdir(parents=True, exist_ok=True)
    rel = f"previews/{uuid.uuid4().hex}.mp4"
    dst = config.STORAGE_DIR / rel
    try:
        make_proxy(src, dst, duration, ctx.progress)
    except BaseException:
        _unlink(dst)
        raise

    old = None
    with SessionLocal() as db:
        v = db.get(Video, video_id)
        if v is None:
            _unlink(dst)
            raise RuntimeError("Video was deleted while building preview")
        old, v.proxy_path = v.proxy_path, rel
        db.commit()
    if old:
        try:
            _unlink(resolve_storage(old))
        except ValueError:
            pass

"""SQLite-backed job queue with background worker threads (default 1 worker for 8GB RAM)."""
import json
import logging
import os
import threading
import time

from sqlalchemy import select, update

from backend.database.db import SessionLocal
from backend.database.models import Clip, Job, utcnow

log = logging.getLogger("clipforge.runner")

HANDLERS = {}
_stop = threading.Event()
_threads: list = []


class JobCancelled(Exception):
    pass


def register(job_type: str):
    def deco(fn):
        HANDLERS[job_type] = fn
        return fn
    return deco


class JobContext:
    def __init__(self, job_id: int, payload: dict):
        self.job_id = job_id
        self.payload = payload
        self._t = 0.0

    def progress(self, p: float) -> None:
        """p is 0.0..1.0. Writes to DB at most once per second and checks for cancellation."""
        now = time.monotonic()
        if now - self._t < 1.0 and p < 1.0:
            return
        self._t = now
        with SessionLocal() as db:
            status = db.scalar(select(Job.status).where(Job.id == self.job_id))
            if status == "cancelled":
                raise JobCancelled()
            db.execute(
                update(Job)
                .where(Job.id == self.job_id, Job.status == "processing")
                .values(progress=round(min(max(p, 0.0), 1.0) * 100, 1), updated_at=utcnow())
            )
            db.commit()


def _claim():
    with SessionLocal() as db:
        row = db.execute(
            select(Job.id, Job.type, Job.payload).where(Job.status == "queued").order_by(Job.id).limit(1)
        ).first()
        if row is None:
            return None
        res = db.execute(
            update(Job).where(Job.id == row.id, Job.status == "queued")
            .values(status="processing", progress=0.0, updated_at=utcnow())
        )
        db.commit()
        return row if res.rowcount == 1 else None


def _finish(job_id: int, status: str, error: str = None) -> None:
    vals = {"status": status, "updated_at": utcnow()}
    if status == "completed":
        vals["progress"] = 100.0
    if error:
        vals["error"] = error[:1000]
    with SessionLocal() as db:
        db.execute(update(Job).where(Job.id == job_id, Job.status == "processing").values(**vals))
        db.commit()


def _worker(_idx: int) -> None:
    while not _stop.is_set():
        try:
            row = _claim()
        except Exception:
            log.exception("could not claim a job")
            _stop.wait(2)
            continue
        if row is None:
            _stop.wait(0.7)
            continue
        handler = HANDLERS.get(row.type)
        if handler is None:
            _finish(row.id, "failed", f"Unknown job type: {row.type}")
            continue
        try:
            payload = json.loads(row.payload) if row.payload else {}
            handler(JobContext(row.id, payload))
            _finish(row.id, "completed")
        except JobCancelled:
            pass  # status already set to cancelled by the API
        except Exception as e:
            log.exception("job %s failed", row.id)
            _finish(row.id, "failed", str(e) or e.__class__.__name__)


def _recover_interrupted() -> None:
    with SessionLocal() as db:
        db.execute(update(Job).where(Job.status == "processing")
                   .values(status="failed", error="Interrupted (server restarted)", updated_at=utcnow()))
        db.execute(update(Clip).where(Clip.status == "processing").values(status="failed"))
        db.commit()


def start() -> None:
    global _threads
    from backend.workers import handlers  # noqa: F401  (registers job handlers)
    if any(t.is_alive() for t in _threads):
        return
    _recover_interrupted()
    _stop.clear()
    n = min(max(int(os.getenv("MAX_CONCURRENT_JOBS", "1") or 1), 1), 4)
    _threads = [threading.Thread(target=_worker, args=(i,), name=f"clipforge-worker-{i}", daemon=True)
                for i in range(n)]
    for t in _threads:
        t.start()


def stop() -> None:
    _stop.set()
    for t in _threads:
        t.join(timeout=5)
