router = APIRouter(prefix="/api/videos", tags=["videos"])
CHUNK = 1024 * 1024


class UploadRejected(Exception):
    def __init__(self, code: str, message: str):
        self.code = code
        self.message = message


def video_dict(v: Video) -> dict:
    return {
        "id": v.id,
        "name": v.original_name,
        "size_bytes": v.size_bytes,
        "duration_sec": v.duration_sec,
        "width": v.width,
        "height": v.height,
        "format": v.format,
        "status": v.status,
        "has_video": v.width is not None,
        "created_at": v.created_at.isoformat(),
    }


def _save_one(upload: UploadFile, db: Session) -> dict:
    display = config.safe_display_name(upload.filename or "")
    ext = Path(display).suffix.lower()
    if ext not in config.ALLOWED_EXTENSIONS:
        allowed = ", ".join(sorted(config.ALLOWED_EXTENSIONS))
        raise UploadRejected("bad_type", f"Unsupported file type '{ext or 'none'}'. Allowed: {allowed}")

    tmp = config.STORAGE_DIR / "temporary" / f"{uuid.uuid4().hex}.part"
    limit = config.max_upload_bytes()
    size = 0
    try:
        with open(tmp, "wb") as out:
            while True:
                chunk = upload.file.read(CHUNK)
                if not chunk:
                    break
                size += len(chunk)
                if size > limit:
                    raise UploadRejected("too_large", "File is larger than the allowed upload limit (MAX_UPLOAD_GB in .env)")
                out.write(chunk)
        if size == 0:
            raise UploadRejected("empty", "File is empty")
        try:
            info = probe(tmp)
        except ProbeError as e:
            raise UploadRejected("invalid_media", str(e))

        final_name = f"{uuid.uuid4().hex}{ext}"
        final = config.STORAGE_DIR / "original_videos" / final_name
        shutil.move(str(tmp), str(final))
    finally:
        tmp.unlink(missing_ok=True)

    try:
        v = Video(
            original_name=display,
            stored_path=f"original_videos/{final_name}",
            size_bytes=size,
            duration_sec=info["duration_sec"],
            width=info["width"],
            height=info["height"],
            format=ext.lstrip("."),
            status="uploaded",
        )
        db.add(v)
        db.commit()
        db.refresh(v)
    except Exception:
        db.rollback()
        final.unlink(missing_ok=True)
        raise
    return video_dict(v)


@router.post("/upload")
def upload_videos(files: list[UploadFile] = File(...), db: Session = Depends(get_db)):
    uploaded, errors = [], []
    for f in files:
        try:
            uploaded.append(_save_one(f, db))
        except UploadRejected as e:
            errors.append({"filename": f.filename, "code": e.code, "error": e.message})
    if uploaded or not errors:
        status = 200
    elif any(e["code"] == "too_large" for e in errors):
        status = 413
    else:
        status = 400
    return JSONResponse({"uploaded": uploaded, "errors": errors}, status_code=status)


@router.get("")
def list_videos(q: str = "", format: str = "", limit: int = 100, offset: int = 0, db: Session = Depends(get_db)):
    stmt = select(Video)
    if q.strip():
        stmt = stmt.where(Video.original_name.contains(q.strip(), autoescape=True))
    if format:
        stmt = stmt.where(Video.format == format.lower().lstrip("."))
    total = db.scalar(select(func.count()).select_from(stmt.subquery())) or 0
    rows = db.scalars(
        stmt.order_by(Video.created_at.desc(), Video.id.desc())
        .limit(min(max(limit, 1), 500))
        .offset(max(offset, 0))
    ).all()
    return {"total": total, "items": [video_dict(v) for v in rows]}


def _get_or_404(db: Session, video_id: int) -> Video:
    v = db.get(Video, video_id)
    if not v:
        raise HTTPException(404, "Video not found")
    return v


def _resolve(v: Video) -> Path:
    base = config.STORAGE_DIR.resolve()
    p = (base / v.stored_path).resolve()
    if base not in p.parents:
        raise HTTPException(400, "Invalid stored path")
    return p

    v = _get_or_404(db, video_id)
    p = _resolve(v)
    if not p.is_file():
        raise HTTPException(404, "File is missing on disk")
    return FileResponse(p, media_type=config.MEDIA_TYPES.get(p.suffix.lower(), "application/octet-stream"))


@router.delete("/{video_id}")

    db.commit()
    return {"deleted": True, "id": video_id}

def _proxy_file(v: Video):
    if not v.proxy_path:
        return None
    try:
        p = resolve_storage(v.proxy_path)
    except ValueError:
        return None
    return p if p.is_file() else None


@router.post("/{video_id}/proxy")
def request_proxy(video_id: int, db: Session = Depends(get_db)):
    """Make a browser-friendly preview copy (used when the original cannot play in the browser)."""
    v = db.get(Video, video_id)
    if not v:
        raise HTTPException(404, "Video not found")
    if v.width is None:
        raise HTTPException(400, "Audio files do not need a video preview")
    if _proxy_file(v):
        return {"ready": True, "job_id": None}
    active = db.scalars(
        select(Job).where(Job.type == "proxy", Job.video_id == video_id, Job.status.in_(["queued", "processing"]))
    ).first()
    if active:
        return {"ready": False, "job_id": active.id}
    job = Job(type="proxy", video_id=video_id, status="queued", payload=json.dumps({"video_id": video_id}))
    db.add(job)
    db.commit()
    db.refresh(job)
    return {"ready": False, "job_id": job.id}


@router.get("/{video_id}/proxy")
def get_proxy(video_id: int, db: Session = Depends(get_db)):
    v = db.get(Video, video_id)
    if not v:
        raise HTTPException(404, "Video not found")
    p = _proxy_file(v)
    if not p:
        raise HTTPException(404, "Preview is not ready")
    return FileResponse(p, media_type="video/mp4")

import json
from typing import Literal, Optional

from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, Field

from sqlalchemy.orm import Session

router = APIRouter(prefix="/api/videos", tags=["auto-clips"])


class AutoClipsBody(BaseModel):
    count: int = Field(5, ge=1, le=20)
    min_score: float = Field(0.0, ge=0, le=100)
    mode: Literal["accurate", "fast"] = "accurate"
    pad_before: float = Field(0.0, ge=0, le=5)
    pad_after: float = Field(0.0, ge=0, le=5)


@router.post("/{video_id}/auto-clips")
def auto_clips(video_id: int, body: Optional[AutoClipsBody] = None, db: Session = Depends(get_db)):
    body = body or AutoClipsBody()
    video = db.get(Video, video_id)
    if video is None:
        raise HTTPException(404, "Video not found")
    analysis = db.scalar(select(VideoAnalysis).where(VideoAnalysis.video_id == video_id))
    if analysis is None or analysis.status != "ready":
        raise HTTPException(409, "Run AI analysis on this video first")
    try:
        highlights = json.loads(analysis.highlights_json or "[]")
    except ValueError:
        highlights = []
    if not highlights:
        raise HTTPException(422, "No highlights found (video may have no speech)")

    existing = db.scalars(
        select(Clip).where(Clip.video_id == video_id, Clip.status.in_(["pending", "processing", "ready"]))
    ).all()
    chosen = select_clips(
        highlights,
        count=body.count,
        min_score=body.min_score,
        duration=video.duration_sec,
        pad_before=body.pad_before,
        pad_after=body.pad_after,
        blocked=[(c.start_sec, c.end_sec) for c in existing],
    )
    if not chosen:
        raise HTTPException(422, "No new clips available (all good highlights already used or score too low)")

    created = []
    for c in chosen:
        clip = Clip(
            video_id=video_id,
            title=c["title"],
            start_sec=c["start"],
            end_sec=c["end"],
            duration_sec=c["duration"],
            status="pending",
            ai_score=c["score"],
        )
        db.add(clip)
        db.flush()
        job = Job(
            type="clip_export",
            video_id=video_id,
            status="queued",
            payload=json.dumps({"clip_id": clip.id, "mode": body.mode}),
        )
        db.add(job)
        db.flush()
        created.append({"clip_id": clip.id, "job_id": job.id, "title": clip.title,
                        "start": c["start"], "end": c["end"], "score": c["score"]})
    db.commit()
    return {"video_id": video_id, "created": len(created), "clips": created}

@router.post("/videos/{video_id}/run")
def run_analysis(video_id: int, body: Optional[RunBody] = None, db: Session = Depends(get_db)):
    body = body or RunBody()
    v = _video_or_404(db, video_id)
    if not body.steps:
        raise HTTPException(400, "Choose at least one step")
    active = _active_job(db, video_id)
    if active:
        return {"job_id": active.id, "already_running": True}
    if "highlights" in body.steps and "transcribe" not in body.steps and _count(db, Transcript, video_id) == 0:
        raise HTTPException(400, "Highlights need a transcript. Include the 'transcribe' step.")
    if body.steps == ["scenes"] and v.width is None:
        raise HTTPException(400, "Scene detection needs a video file (this file is audio only)")
    job = Job(type="analyze", video_id=video_id, status="queued",
              payload=json.dumps({"video_id": video_id, "steps": body.steps,
                                  "use_gemini": body.use_gemini, "force": body.force}))
    db.add(job)
    db.commit()
    db.refresh(job)
    return {"job_id": job.id, "already_running": False}

import uuid

from backend import config

def _aspect(w, h):
    if not w or not h:
        return None
    r = w / h
    for name, val in (("16:9", 16 / 9), ("9:16", 9 / 16), ("1:1", 1.0), ("4:3", 4 / 3), ("3:4", 3 / 4), ("21:9", 21 / 9)):
        if abs(r - val) < 0.03:
            return name
    return "custom"


def _set_clip(clip_id: int, **values) -> bool:
    with SessionLocal() as db:
        clip = db.get(Clip, clip_id)
        if clip is None:
            return False
        for k, v in values.items():
            setattr(clip, k, v)
        db.commit()
        return True


def _unlink(*paths) -> None:
    for p in paths:
        if p is not None:
            try:
                p.unlink(missing_ok=True)
            except OSError:
                pass


@register("clip_export")
def clip_export(ctx: JobContext) -> None:
    clip_id = int(ctx.payload["clip_id"])
    mode = ctx.payload.get("mode", "accurate")
    with SessionLocal() as db:
        clip = db.get(Clip, clip_id)
        if clip is None:
            raise RuntimeError("Clip record not found")
        video = db.get(Video, clip.video_id)
        if video is None:
            raise RuntimeError("Source video not found")
        stored = video.stored_path
        start, end = clip.start_sec, clip.end_sec
        clip.status = "processing"
        db.commit()

    src = resolve_storage(stored)
    if not src.is_file():
        _set_clip(clip_id, status="failed")
        raise RuntimeError("Source video file is missing on disk")

    name = uuid.uuid4().hex
    dst = config.STORAGE_DIR / "generated_clips" / f"{name}.mp4"
    thumb = config.STORAGE_DIR / "thumbnails" / f"{name}.jpg"
    try:
        if mode == "fast":
            try:
                cut_clip(src, dst, start, end, "fast", ctx.progress)
            except FfmpegError:  # codec not allowed in MP4 container: fall back to re-encode
                _unlink(dst)
                cut_clip(src, dst, start, end, "accurate", ctx.progress)
        else:
            cut_clip(src, dst, start, end, "accurate", ctx.progress)

        info = probe(dst)
        thumb_ok = True
        try:
            grab_frame(dst, thumb, min(1.0, (end - start) / 2))
        except FfmpegError:
            thumb_ok = False

        ok = _set_clip(
            clip_id,
            status="ready",
            file_path=f"generated_clips/{name}.mp4",
            thumbnail_path=f"thumbnails/{name}.jpg" if thumb_ok else None,
            width=info["width"],
            height=info["height"],
            aspect_ratio=_aspect(info["width"], info["height"]),
            duration_sec=info["duration_sec"] or (end - start),
        )
        if not ok:
            raise RuntimeError("Clip was deleted while exporting")
    except JobCancelled:
        _unlink(dst, thumb)
        _set_clip(clip_id, status="cancelled")
        raise
    except Exception:
        _unlink(dst, thumb)
        _set_clip(clip_id, status="failed")
        raise


@register("proxy")
def build_proxy(ctx: JobContext) -> None:
    video_id = int(ctx.payload["video_id"])
    with SessionLocal() as db:
        v = db.get(Video, video_id)
        if v is None:
            raise RuntimeError("Video not found")
        stored, duration = v.stored_path, (v.duration_sec or 0.0)
    src = resolve_storage(stored)
    if not src.is_file():
        raise RuntimeError("Source video file is missing on disk")

    (config.STORAGE_DIR / "previews").mkdir(parents=True, exist_ok=True)
    rel = f"previews/{uuid.uuid4().hex}.mp4"
    dst = config.STORAGE_DIR / rel
    try:
        make_proxy(src, dst, duration, ctx.progress)
    except BaseException:
        _unlink(dst)
        raise

    old = None
    with SessionLocal() as db:
        v = db.get(Video, video_id)
        if v is None:
            _unlink(dst)
            raise RuntimeError("Video was deleted while building preview")
        old, v.proxy_path = v.proxy_path, rel
        db.commit()
    if old:
        try:
            _unlink(resolve_storage(old))
        except ValueError:
            pass

def _sub(ctx: JobContext, lo: float, hi: float):
    def cb(p: float) -> None:
        ctx.progress(lo + (hi - lo) * min(max(p, 0.0), 1.0))
    return cb


def _upsert_va(video_id: int, **values) -> None:
    with SessionLocal() as db:
        row = db.scalar(select(VideoAnalysis).where(VideoAnalysis.video_id == video_id))
        if row is None:
            row = VideoAnalysis(video_id=video_id)
            db.add(row)
        for k, v in values.items():
            setattr(row, k, v)
        row.updated_at = utcnow()
        db.commit()


def _safe_upsert_va(video_id: int, **values) -> None:
    try:
        _upsert_va(video_id, **values)
    except Exception:
        log.exception("could not update video_analysis for video %s", video_id)


def _sync_va(video_id: int) -> None:
    """Rebuild the video_analysis row from the normalized tables."""
    with SessionLocal() as db:
        tr = db.scalars(select(Transcript).where(Transcript.video_id == video_id)
                        .order_by(Transcript.id.desc())).first()
        if tr is None:
            return
        language = tr.language
        segments_json = tr.segments or "[]"
        segments = json.loads(segments_json)
        scenes = [{"index": s.idx, "start": s.start_sec, "end": s.end_sec}
                  for s in db.scalars(select(Scene).where(Scene.video_id == video_id).order_by(Scene.idx)).all()]
        rows = db.scalars(select(Highlight).where(Highlight.video_id == video_id)
                          .order_by(Highlight.score.desc(), Highlight.id)).all()
        highlights = [
            {"rank": i, "start": h.start_sec, "end": h.end_sec, "duration": round(h.end_sec - h.start_sec, 2),
             "score": h.score, "local_score": h.local_score, "title": h.title, "reason": h.reason,
             "text": h.text, "source": h.source}
            for i, h in enumerate(rows, 1)
        ]
        row = db.get(Setting, info_key(video_id))
        info = json.loads(row.value) if row else {}
    source = "gemini" if info.get("gemini_used") else "local"
    note = "; ".join(info.get("warnings") or [])
    if not segments:
        note = "No speech detected - nothing to score."
    _upsert_va(
        video_id, status="ready", error=None, language=language, transcript_json=segments_json,
        scenes_json=json.dumps(scenes), highlights_json=json.dumps(highlights, ensure_ascii=False),
        scoring_source=source, note=(note[:500] or None),
    )


def _run_steps(ctx, video_id, src, duration, has_video, have_transcript, steps, use_gemini, force, language):
    todo = []
    for s in steps:
        if s == "scenes" and not has_video:
            continue
        if s == "transcribe" and have_transcript and not force:
            continue
        todo.append(s)
    if "highlights" in todo and "transcribe" not in todo and not have_transcript:
        raise RuntimeError("Highlights need a transcript. Run the transcribe step first.")

    total = sum(WEIGHTS[s] for s in todo) or 1.0
    done = 0.0
    for s in todo:
        cb = _sub(ctx, done / total, min((done + WEIGHTS[s]) / total, 0.99))

        if s == "transcribe":
            kwargs = {"language": language} if language else {}
            result = transcribe_media(src, duration, cb, **kwargs)
            with SessionLocal() as db:
                db.execute(delete(Transcript).where(Transcript.video_id == video_id))
                db.add(Transcript(
                    video_id=video_id,
                    language=result.get("language"),
                    language_probability=result.get("language_probability"),
                    duration_sec=result.get("duration_sec") or duration,
                    text=result.get("text", ""),
                    segments=json.dumps(result.get("segments", []), ensure_ascii=False),
                    model=result.get("model"),
                ))
                db.commit()

        elif s == "scenes":
            scenes = detect_scenes(src, duration, cb)
            with SessionLocal() as db:
                db.execute(delete(Scene).where(Scene.video_id == video_id))
                for sc in scenes:
                    db.add(Scene(video_id=video_id, idx=sc["idx"], start_sec=sc["start"], end_sec=sc["end"]))
                db.commit()

        elif s == "highlights":
            with SessionLocal() as db:
                tr = db.scalars(select(Transcript).where(Transcript.video_id == video_id)
                                .order_by(Transcript.id.desc())).first()
                segments = json.loads(tr.segments) if tr else []
            items, info = analyze_highlights(segments, use_gemini=use_gemini, on_progress=cb)
            with SessionLocal() as db:
                db.execute(delete(Highlight).where(Highlight.video_id == video_id))
                for h in items:
                    db.add(Highlight(
                        video_id=video_id, start_sec=h["start"], end_sec=h["end"], score=h["score"],
                        local_score=h["local_score"], title=h["title"][:300], reason=h["reason"],
                        text=h["text"], source=h["source"],
                    ))
                row = db.get(Setting, info_key(video_id))
                payload = json.dumps(info, ensure_ascii=False)
                if row:
                    row.value = payload
                else:
                    db.add(Setting(key=info_key(video_id), value=payload))
                db.commit()
            for w in info["warnings"]:
                log.warning("video %s: %s", video_id, w)
        done += WEIGHTS[s]


@register("analyze")
def analyze(ctx: JobContext) -> None:
    video_id = int(ctx.payload["video_id"])
    wanted = ctx.payload.get("steps") or list(ORDER)
    steps = [s for s in ORDER if s in wanted]
    use_gemini = bool(ctx.payload.get("use_gemini", True))
    force = bool(ctx.payload.get("force", False))
    language = ctx.payload.get("language") or None
    if language == "auto":
        language = None

    with SessionLocal() as db:
        v = db.get(Video, video_id)
        if v is None:
            raise RuntimeError("Video not found")
        stored, duration, has_video = v.stored_path, float(v.duration_sec or 0.0), v.width is not None
        have_transcript = (db.scalar(
            select(func.count()).select_from(Transcript).where(Transcript.video_id == video_id)) or 0) > 0

    src = resolve_storage(stored)
    if not src.is_file():
        raise RuntimeError("Source video file is missing on disk")

    touches_va = any(s in steps for s in ("transcribe", "highlights"))
    if touches_va:
        _safe_upsert_va(video_id, status="processing", error=None)
    try:
        _run_steps(ctx, video_id, src, duration, has_video, have_transcript, steps, use_gemini, force, language)
        _sync_va(video_id)
    except JobCancelled:
        if touches_va:
            _safe_upsert_va(video_id, status="cancelled")
        raise
    except Exception as e:
        if touches_va:
            _safe_upsert_va(video_id, status="failed", error=(str(e) or e.__class__.__name__)[:500])
        raise

import json
import shutil
import subprocess


class ProbeError(Exception):
    pass


def probe(path) -> dict:
    exe = shutil.which("ffprobe")
    if not exe:
        raise ProbeError("ffprobe not found. Install FFmpeg and restart VS Code.")
    try:
        r = subprocess.run(
            [exe, "-v", "error", "-print_format", "json", "-show_format", "-show_streams", str(path)],
            capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=60,
        )
    except subprocess.TimeoutExpired:
        raise ProbeError("ffprobe timed out while reading the file")
    if r.returncode != 0:
        raise ProbeError("This is not a valid audio/video file")
    try:
        data = json.loads(r.stdout or "{}")
    except json.JSONDecodeError:
        raise ProbeError("Could not read media information")

    streams = data.get("streams", [])
    video = next(
        (s for s in streams
         if s.get("codec_type") == "video" and s.get("disposition", {}).get("attached_pic", 0) != 1),
        None,
    )
    audio = next((s for s in streams if s.get("codec_type") == "audio"), None)
    if not video and not audio:
        raise ProbeError("No audio or video stream found in this file")

    raw = data.get("format", {}).get("duration") or (video or audio).get("duration")
    try:
        duration = float(raw) if raw is not None else None
    except (TypeError, ValueError):
        duration = None

    return {
        "duration_sec": duration,
        "width": video.get("width") if video else None,
        "height": video.get("height") if video else None,
    }

"""Burn an ASS subtitle file into a video (needs an FFmpeg build with libass)."""
import collections
import re
import subprocess
from pathlib import Path
from typing import Callable, Optional

from backend.video_processing.ffmpeg_service import _NO_WINDOW, _PROGRESS_KEYS, FfmpegError, _exe

ProgressCb = Optional[Callable[[float], None]]
_CACHE = {}


def has_subtitles_filter() -> bool:
    if "v" not in _CACHE:
        try:
            r = subprocess.run([_exe(), "-hide_banner", "-filters"], capture_output=True, text=True,
                               encoding="utf-8", errors="replace", timeout=30, creationflags=_NO_WINDOW)
            _CACHE["v"] = bool(re.search(r"(?m)^\s*[A-Z.]{3}\s+subtitles\s", r.stdout))
        except Exception:
            _CACHE["v"] = False
    return _CACHE["v"]


def burn_captions(src, ass_path, dst, total_sec: float = 0.0, on_progress: ProgressCb = None) -> None:
    """Re-encode video with the captions drawn in. Audio is copied. Original is never modified.
    FFmpeg runs inside the .ass folder, so the filter gets a plain file name (no Windows path escaping)."""
    src, ass_path, dst = Path(src).resolve(), Path(ass_path).resolve(), Path(dst).resolve()
    cmd = [
        _exe(), "-hide_banner", "-nostdin", "-y", "-progress", "pipe:1", "-nostats",
        "-i", str(src), "-map", "0:v:0", "-map", "0:a:0?",
        "-vf", f"subtitles={ass_path.name}",
        "-c:v", "libx264", "-preset", "veryfast", "-crf", "20", "-pix_fmt", "yuv420p",
        "-c:a", "copy", "-movflags", "+faststart", str(dst),
    ]
    tail = collections.deque(maxlen=25)
    proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True,
                            encoding="utf-8", errors="replace", creationflags=_NO_WINDOW, cwd=str(ass_path.parent))
    try:
        assert proc.stdout is not None
        for raw in proc.stdout:
            line = raw.strip()
            if not line:
                continue
            if line.startswith(("out_time_us=", "out_time_ms=")):
                if on_progress and total_sec > 0:
                    try:
                        us = int(line.split("=", 1)[1])
                    except ValueError:
                        continue
                    on_progress(min(max(us / 1_000_000 / total_sec, 0.0), 1.0))
            elif not _PROGRESS_KEYS.match(line):
                tail.append(line)
        code = proc.wait()
    finally:
        if proc.poll() is None:
            proc.kill()
            proc.wait()
    if code != 0:
        msg = "\n".join(tail)[-800:] or f"ffmpeg exited with code {code}"
        if "subtitles" in msg and "No such filter" in msg:
            msg = "This FFmpeg build has no subtitles (libass) support. " + msg
        raise FfmpegError(msg)
