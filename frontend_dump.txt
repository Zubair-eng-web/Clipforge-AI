function fmtBytes(n: number) {
  if (n < 1024) return `${n} B`;
  const units = ["KB", "MB", "GB", "TB"];
  let i = -1;
  do { n /= 1024; i++; } while (n >= 1024 && i < units.length - 1);
  return `${n.toFixed(1)} ${units[i]}`;
}

import { useCallback, useEffect, useRef, useState } from "react";
import { Loader2, Scissors, Sparkles } from "lucide-react";

const API = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000";

type VideoItem = { id: number; name: string };
type RawVideo = { id: number; original_name?: string; name?: string; title?: string };
type Segment = { start: number; end: number; text: string };
type Scene = { index: number; start: number; end: number };
type Highlight = {
  rank: number;
  start: number;
  end: number;
  duration: number;
  score: number;
  local_score: number;
  title: string;
  reason: string;
  text: string;
  source: string;
};
type Analysis = {
  status: string;
  language?: string | null;
  scoring_source?: string | null;
  note?: string | null;
  error?: string | null;
  segment_count?: number;
  scenes?: Scene[];
  highlights?: Highlight[];
  transcript?: Segment[];
};

const fmt = (sec: number) => {
  const s = Math.max(0, Math.floor(sec));
  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, "0")}`;
};

const JSON_HEADERS = { "Content-Type": "application/json" };

export default function AiHighlightsPage() {
  const [videos, setVideos] = useState<VideoItem[]>([]);
  const [videoId, setVideoId] = useState<number | null>(null);
  const [analysis, setAnalysis] = useState<Analysis | null>(null);
  const [progress, setProgress] = useState<number | null>(null);
  const [msg, setMsg] = useState("");
  const [showTranscript, setShowTranscript] = useState(false);
  const esRef = useRef<EventSource | null>(null);

  useEffect(() => {
    fetch(`${API}/api/videos`)
      .then((r) => r.json())
      .then((data) => {
        const list: RawVideo[] = Array.isArray(data) ? data : data.items ?? data.videos ?? [];
        setVideos(
          list.map((v) => ({
            id: v.id,
            name: v.original_name ?? v.name ?? v.title ?? `Video ${v.id}`,
          }))
        );
      })
      .catch(() => setMsg("Cannot reach the backend at " + API));
    return () => esRef.current?.close();
  }, []);

  const loadAnalysis = useCallback(async (id: number) => {
    try {
      const r = await fetch(`${API}/api/videos/${id}/analysis`);
      setAnalysis(r.ok ? await r.json() : null);
    } catch {
      setAnalysis(null);
    }
  }, []);

  function pick(id: number | null) {
    esRef.current?.close();
    setProgress(null);
    setMsg("");
    setVideoId(id);
    setAnalysis(null);
    setShowTranscript(false);
    if (id !== null) loadAnalysis(id);
  }

  async function startAnalysis() {
    if (videoId === null) return;
    const id = videoId;
    setMsg("");
    const r = await fetch(`${API}/api/videos/${id}/analyze`, {
      method: "POST",
      headers: JSON_HEADERS,
      body: JSON.stringify({ language: null, use_gemini: true }),
    });
    const data = await r.json().catch(() => ({}));
    if (!r.ok) {
      setMsg(typeof data.detail === "string" ? data.detail : "Could not start the analysis.");
      return;
    }
    setProgress(0);
    const es = new EventSource(`${API}/api/jobs/${data.job_id}/events`);
    esRef.current = es;
    es.onmessage = (e) => {
      const d = JSON.parse(e.data);
      if (typeof d.progress === "number") setProgress(d.progress);
      if (["completed", "failed", "cancelled"].includes(d.status)) {
        es.close();
        setProgress(null);
        if (d.status !== "completed") setMsg(d.error || `Job ${d.status}`);
        loadAnalysis(id);
      }
    };
    es.onerror = () => {
      es.close();
      setProgress(null);
      loadAnalysis(id);
    };
  }

  async function exportClip(h: Highlight) {
    if (videoId === null) return;
    const r = await fetch(`${API}/api/clips`, {
      method: "POST",
      headers: JSON_HEADERS,
      body: JSON.stringify({
        video_id: videoId,
        start_sec: h.start,
        end_sec: h.end,
        title: h.title || `Highlight ${h.rank}`,
        mode: "accurate",
      }),
    });
    const data = await r.json().catch(() => ({}));
    setMsg(
      r.ok
        ? `Highlight #${h.rank} queued for export. Check Generated Clips.`
        : typeof data.detail === "string"
        ? data.detail
        : "Clip export failed."
    );
  }

  const busy = progress !== null;
  const highlights = analysis?.highlights ?? [];

  return (
    <div className="mx-auto max-w-5xl space-y-6 p-6 text-zinc-100">
      <div>
        <h1 className="flex items-center gap-2 text-2xl font-bold">
          <Sparkles className="h-6 w-6 text-violet-400" /> AI Highlights
        </h1>
        <p className="text-sm text-zinc-400">
          Transcript, scenes and AI scores to find the best moments of a video.
        </p>
      </div>

      <div className="flex flex-wrap items-center gap-3">
        <select
          value={videoId ?? ""}
          onChange={(e) => pick(e.target.value ? Number(e.target.value) : null)}
          className="min-w-64 rounded-lg border border-zinc-700 bg-zinc-900 px-3 py-2 text-sm"
        >
          <option value="">Select a video...</option>
          {videos.map((v) => (
            <option key={v.id} value={v.id}>
              {v.name}
            </option>
          ))}
        </select>
        <button
          onClick={startAnalysis}
          disabled={videoId === null || busy}
          className="inline-flex items-center gap-2 rounded-lg bg-violet-600 px-4 py-2 text-sm font-medium hover:bg-violet-500 disabled:opacity-50"
        >
          {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Sparkles className="h-4 w-4" />}
          {analysis?.status === "ready" ? "Analyze again" : "Analyze"}
        </button>
      </div>

      {busy && (
        <div>
          <div className="h-2 w-full overflow-hidden rounded bg-zinc-800">
            <div
              className="h-2 bg-violet-500 transition-all"
              style={{ width: `${Math.min(progress ?? 0, 100)}%` }}
            />
          </div>
          <p className="mt-1 text-xs text-zinc-400">
            Analyzing... {Math.round(progress ?? 0)}% (first run downloads the Whisper model)
          </p>
        </div>
      )}

      {msg && <p className="rounded-lg bg-zinc-900 p-3 text-sm text-amber-300">{msg}</p>}

      {analysis?.status === "failed" && (
        <p className="rounded-lg bg-zinc-900 p-3 text-sm text-red-400">
          Last analysis failed: {analysis.error}
        </p>
      )}

      {analysis?.status === "ready" && (
        <div className="space-y-4">
          <p className="text-sm text-zinc-400">
            Language: {analysis.language ?? "?"} | {analysis.segment_count ?? 0} segments |{" "}
            {analysis.scenes?.length ?? 0} scenes | scoring: {analysis.scoring_source}
          </p>
          {analysis.note && <p className="text-sm text-amber-300">{analysis.note}</p>}
          {highlights.length === 0 && (
            <p className="text-sm text-zinc-400">No highlights found for this video.</p>
          )}
          {highlights.map((h) => (
            <div key={h.rank} className="rounded-xl border border-zinc-800 bg-zinc-900 p-4">
              <div className="flex items-start justify-between gap-3">
                <div>
                  <div className="text-xs text-zinc-500">
                    #{h.rank} | {fmt(h.start)} - {fmt(h.end)} ({Math.round(h.duration)}s)
                  </div>
                  <div className="font-semibold">{h.title}</div>
                  <div className="text-xs text-zinc-400">{h.reason}</div>
                </div>
                <div className="text-right">
                  <div className="text-2xl font-bold text-violet-400">{Math.round(h.score)}</div>
                  <div className="text-[10px] uppercase text-zinc-500">{h.source}</div>
                </div>
              </div>
              <p className="mt-2 text-sm text-zinc-300">{h.text}</p>
              <button
                onClick={() => exportClip(h)}
                className="mt-3 inline-flex items-center gap-1 rounded-lg bg-violet-600 px-3 py-1.5 text-sm hover:bg-violet-500"
              >
                <Scissors className="h-4 w-4" /> Export clip
              </button>
            </div>
          ))}

          <button
            onClick={() => setShowTranscript((s) => !s)}
            className="text-sm text-violet-400 hover:underline"
          >
            {showTranscript ? "Hide transcript" : "Show transcript"}
          </button>
          {showTranscript && (
            <div className="max-h-96 space-y-1 overflow-y-auto rounded-xl border border-zinc-800 bg-zinc-900 p-4 text-sm">
              {(analysis.transcript ?? []).map((s, i) => (
                <p key={i}>
                  <span className="mr-2 text-xs text-zinc-500">{fmt(s.start)}</span>

"use client";

import { useCallback, useEffect, useState } from "react";
import { Download, Loader2, Play, Sparkles, Wand2 } from "lucide-react";

const API = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000";

type VideoItem = { id: number; name: string };
type ClipItem = {
  id: number;
  title: string;
  start_sec: number;
  end_sec: number;
  status: string;
  ai_score?: number | null;
};

function asList(d: any): any[] {
  if (Array.isArray(d)) return d;
  return d?.items ?? d?.clips ?? d?.videos ?? [];
}

function fmt(sec: number) {
  const s = Math.max(0, Math.round(sec));
  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, "0")}`;
}

async function api(path: string, init?: RequestInit) {
  const r = await fetch(`${API}${path}`, init);
  const data = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error(typeof data?.detail === "string" ? data.detail : `Request failed (${r.status})`);
  return data;
}

export default function AutoClipperPage() {
  const [videos, setVideos] = useState<VideoItem[]>([]);
  const [videoId, setVideoId] = useState<number | null>(null);
  const [analysisStatus, setAnalysisStatus] = useState("none");
  const [highlightCount, setHighlightCount] = useState(0);
  const [source, setSource] = useState("");
  const [note, setNote] = useState("");
  const [count, setCount] = useState(5);
  const [mode, setMode] = useState<"accurate" | "fast">("accurate");
  const [clips, setClips] = useState<ClipItem[]>([]);
  const [busy, setBusy] = useState("");
  const [error, setError] = useState("");

  useEffect(() => {
    api("/api/videos")
      .then((d) => {
        const list = asList(d).map((v: any) => ({
          id: v.id,
          name: v.original_name ?? v.name ?? v.title ?? `Video ${v.id}`,
        }));
        setVideos(list);
        if (list.length) setVideoId(list[0].id);
      })
      .catch((e) => setError(e.message));
  }, []);

  const loadAnalysis = useCallback(async (id: number) => {
    try {
      const a = await api(`/api/videos/${id}/analysis?include_transcript=false`);
      setAnalysisStatus(a.status || "none");
      setHighlightCount((a.highlights || []).length);
      setSource(a.scoring_source || "");
      setNote(a.note || "");
    } catch (e: any) {
      setError(e.message);
    }
  }, []);

  const loadClips = useCallback(async (id: number) => {
    try {
      const d = await api(`/api/clips?video_id=${id}`);
      setClips(asList(d));
    } catch (e: any) {
      setError(e.message);
    }
  }, []);

  useEffect(() => {
    if (videoId == null) return;
    setError("");
    loadAnalysis(videoId);
    loadClips(videoId);
  }, [videoId, loadAnalysis, loadClips]);

  useEffect(() => {
    if (videoId == null) return;
    const active = clips.some((c) => c.status === "pending" || c.status === "processing");
    if (!active) return;
    const t = setInterval(() => loadClips(videoId), 2000);
    return () => clearInterval(t);
  }, [clips, videoId, loadClips]);

  async function analyze() {
    if (videoId == null) return;
    setError("");
    setBusy("Analyzing (transcript, scenes, scoring)... this can take a few minutes on CPU");
    try {
      const { job_id } = await api(`/api/videos/${videoId}/analyze`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({}),
      });
      for (;;) {
        await new Promise((r) => setTimeout(r, 3000));
        const j = await api(`/api/jobs/${job_id}`);
        if (j.status === "completed") break;
        if (j.status === "failed" || j.status === "cancelled") {
          throw new Error(j.error || `Analysis ${j.status}`);
        }
        setBusy(`Analyzing... ${Math.round(j.progress ?? 0)}%`);
      }
      await loadAnalysis(videoId);
    } catch (e: any) {
      setError(e.message);
    } finally {
      setBusy("");
    }
  }

  async function generate() {
    if (videoId == null) return;
    setError("");
    setBusy("Creating clips...");
    try {
      await api(`/api/videos/${videoId}/auto-clips`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ count, mode }),
      });
      await loadClips(videoId);
    } catch (e: any) {
      setError(e.message);
    } finally {
      setBusy("");
    }
  }

  const ready = analysisStatus === "ready";

  return (
    <div className="p-6 space-y-6 max-w-5xl">
      <div>
        <h1 className="text-2xl font-bold flex items-center gap-2">
          <Wand2 className="w-6 h-6" /> AI Auto Clipper
        </h1>
        <p className="text-sm opacity-70">Pick a video, analyze it, then export the best non-overlapping clips.</p>
      </div>

      {error && <div className="rounded border border-red-500/50 bg-red-500/10 p-3 text-sm">{error}</div>}

      <div className="rounded-lg border border-white/10 p-4 space-y-4">
        <div className="flex flex-wrap items-center gap-3">
          <select
            className="rounded border border-white/10 bg-transparent px-3 py-2"
            value={videoId ?? ""}

import { useRef, useState } from "react";

import { UploadCloud, CheckCircle2, AlertTriangle, Loader2, Clock } from "lucide-react";
import { API, ALLOWED_EXT, fmtBytes, fmtDuration, type VideoItem } from "@/lib/api";

type Item = {
  key: string;
  file: File;
  status: "queued" | "uploading" | "done" | "error";
  progress: number;
  error?: string;
  video?: VideoItem;
};

export default function UploadPage() {
  const [items, setItems] = useState<Item[]>([]);
  const [dragging, setDragging] = useState(false);
  const chain = useRef<Promise<void>>(Promise.resolve());
  const inputRef = useRef<HTMLInputElement>(null);

  const patch = (key: string, p: Partial<Item>) =>
    setItems((prev) => prev.map((i) => (i.key === key ? { ...i, ...p } : i)));

  function uploadOne(item: Item): Promise<void> {
    return new Promise((resolve) => {
      const xhr = new XMLHttpRequest();
      xhr.open("POST", `${API}/api/videos/upload`);
      xhr.upload.onprogress = (e) => {
        if (e.lengthComputable) patch(item.key, { progress: Math.round((e.loaded / e.total) * 100) });
      };
      xhr.onload = () => {
        try {
          const data = JSON.parse(xhr.responseText);
          if (xhr.status >= 200 && xhr.status < 300 && data.uploaded?.length) {
            patch(item.key, { status: "done", progress: 100, video: data.uploaded[0] });
          } else {
            patch(item.key, {
              status: "error",
              error: data.errors?.[0]?.error ?? data.detail ?? `Upload failed (${xhr.status})`,
            });
          }
        } catch {
          patch(item.key, { status: "error", error: `Upload failed (${xhr.status})` });
        }
        resolve();
      };
      xhr.onerror = () => {
        patch(item.key, { status: "error", error: "Cannot reach the backend. Is it running?" });
        resolve();
      };
      const form = new FormData();
      form.append("files", item.file);
      patch(item.key, { status: "uploading", progress: 0 });
      xhr.send(form);
    });
  }

  function addFiles(list: FileList | File[]) {
    const fresh: Item[] = Array.from(list).map((file, idx) => {
      const ext = "." + (file.name.split(".").pop() ?? "").toLowerCase();
      const ok = ALLOWED_EXT.includes(ext);
      return {
        key: `${Date.now()}-${idx}-${file.name}`,
        file,
        status: ok ? "queued" : "error",
        progress: 0,
        error: ok ? undefined : `Unsupported file type "${ext}"`,
      } as Item;
    });
    setItems((prev) => [...fresh, ...prev]);
    for (const it of fresh) {
      if (it.status === "queued") chain.current = chain.current.then(() => uploadOne(it));
    }
  }

  return (
    <div className="max-w-3xl">
      <h1 className="text-2xl font-bold text-white">Upload Videos</h1>
      <p className="text-sm text-zinc-400">
        MP4, MOV, MKV, AVI, WebM, MP3, WAV. Files are saved permanently on this computer.
      </p>

      <div
        onDragOver={(e) => { e.preventDefault(); setDragging(true); }}
        onDragLeave={() => setDragging(false)}
        onDrop={(e) => { e.preventDefault(); setDragging(false); addFiles(e.dataTransfer.files); }}
        onClick={() => inputRef.current?.click()}
        className={`mt-6 cursor-pointer rounded-xl border-2 border-dashed p-12 text-center transition-colors ${
          dragging ? "border-violet-500 bg-violet-600/10" : "border-zinc-700 bg-zinc-900/40 hover:border-zinc-500"

            </div>

              <div className="mt-3 h-1.5 overflow-hidden rounded bg-zinc-800">
                <div className="h-full bg-violet-500 transition-all" style={{ width: `${it.progress}%` }} />
              </div>
            )}
            {it.status === "done" && it.video && (
              <p className="mt-2 text-xs text-zinc-400">
                Duration {fmtDuration(it.video.duration_sec)}
                {it.video.width ? ` | ${it.video.width}x${it.video.height}` : " | audio only"}
              </p>
            )}
            {it.status === "error" && <p className="mt-2 text-xs text-red-400">{it.error}</p>}
          </div>
        ))}
      </div>

      {items.some((i) => i.status === "done") && (
        <Link href="/library" className="mt-6 inline-block rounded-lg bg-violet-600 px-4 py-2 text-sm font-medium text-white hover:bg-violet-500">
          Open Video Library
        </Link>
      )}
    </div>
  );
}

  id: number;
  name: string;
  size_bytes: number;
  duration_sec: number | null;
  width: number | null;
  height: number | null;
  format: string | null;
  status: string;
  has_video: boolean;
  created_at: string;

export function fmtBytes(n: number): string {
  if (n < 1024) return `${n} B`;
  const units = ["KB", "MB", "GB", "TB"];
  let i = -1;
  do { n /= 1024; i++; } while (n >= 1024 && i < units.length - 1);
  return `${n.toFixed(1)} ${units[i]}`;
}

export function fmtDuration(sec: number | null): string {
  if (sec == null) return "-";
  const s = Math.round(sec);
  const h = Math.floor(s / 3600);
  const m = Math.floor((s % 3600) / 60);
  const r = s % 60;
  const pad = (n: number) => String(n).padStart(2, "0");
  return h > 0 ? `${h}:${pad(m)}:${pad(r)}` : `${m}:${pad(r)}`;
}

export type ClipItem = {
  id: number;
  video_id: number;
  title: string;
  start_sec: number;
  end_sec: number;
  duration_sec: number;
  width: number | null;
  height: number | null;
  aspect_ratio: string | null;
  status: string;
  ai_score: number | null;
  has_file: boolean;
  has_thumbnail: boolean;
  created_at: string;
};

export type JobItem = {
  id: number;
  type: string;
  video_id: number | null;
  status: string;
  progress: number; // percent 0..100
  error: string | null;
  created_at: string;
  updated_at: string;
};

export function fmtClock(sec: number): string {
  if (!Number.isFinite(sec) || sec < 0) sec = 0;
  const m = Math.floor(sec / 60);
  const s = sec - m * 60;
  return `${m}:${s.toFixed(1).padStart(4, "0")}`;
}

/** Live job progress via Server-Sent Events. Returns a function that stops listening. */
export function watchJob(jobId: number, onUpdate: (j: JobItem) => void): () => void {
  const es = new EventSource(`${API}/api/jobs/${jobId}/events`);
  es.onmessage = (e) => {
    try {
      const j = JSON.parse(e.data) as JobItem;
      onUpdate(j);
      if (["completed", "failed", "cancelled"].includes(j.status)) es.close();
    } catch {
      /* ignore malformed event */
    }
  };
  es.onerror = () => {
    es.close();
    onUpdate({
      id: jobId, type: "", video_id: null, status: "failed", progress: 0,
      error: "Lost connection to the backend", created_at: "", updated_at: "",
    });
  };
  return () => es.close();
}
