"""Analysis pipeline: transcript -> scenes -> highlight candidates -> scores."""
import time

from backend.ai.highlights import build_candidates, score_candidates

FIELDS = ("start", "end", "duration", "score", "local_score", "title", "reason", "text", "source")


def analyze_video(
    path,
    duration,
    progress,
    language=None,
    use_gemini=True,
    transcribe_fn=None,
    scenes_fn=None,
    ask=None,
    sleep=time.sleep,
) -> dict:
    """Run the full analysis. progress(0..1) is called between stages."""
    if transcribe_fn is None:
        from backend.audio_processing.transcribe import transcribe as transcribe_fn
    if scenes_fn is None:
        from backend.scene_detection.detect import detect_scenes as scenes_fn

    progress(0.02)
    tr = transcribe_fn(path, duration, lambda p: progress(0.02 + 0.68 * p), language)
    segments = tr.get("segments", [])
    progress(0.70)
    scenes = scenes_fn(path)
    progress(0.85)
    cands = build_candidates(segments, scenes)
    scored, source, note = score_candidates(cands, use_gemini, ask=ask, sleep=sleep)
    progress(0.97)
    if not segments:
        note = "No speech detected - nothing to score."
    highlights = [
        {"rank": rank, **{k: c[k] for k in FIELDS}} for rank, c in enumerate(scored, 1)
    ]
    return {
        "language": tr.get("language"),
        "segments": segments,
        "scenes": scenes,
        "highlights": highlights,
        "scoring_source": source,
        "note": note,
    }

"""Job handler: analyze (transcript + scenes + highlight scores)."""
import json

from sqlalchemy import select

from backend.ai.pipeline import analyze_video
from backend.database.db import SessionLocal
from backend.database.models import Video, VideoAnalysis, utcnow
from backend.services.paths import resolve_storage
from backend.workers.runner import JobCancelled, JobContext, register


def _upsert(video_id: int, **values) -> None:
    with SessionLocal() as db:
        row = db.scalar(select(VideoAnalysis).where(VideoAnalysis.video_id == video_id))
        if row is None:
            row = VideoAnalysis(video_id=video_id)
            db.add(row)
        for k, v in values.items():
            setattr(row, k, v)
        row.updated_at = utcnow()
        db.commit()


@register("analyze")
def analyze(ctx: JobContext) -> None:
    video_id = int(ctx.payload["video_id"])
    language = ctx.payload.get("language") or None
    if language == "auto":
        language = None
    use_gemini = bool(ctx.payload.get("use_gemini", True))

    with SessionLocal() as db:
        v = db.get(Video, video_id)
        if v is None:
            raise RuntimeError("Video not found")
        stored, duration = v.stored_path, (v.duration_sec or 0.0)
    src = resolve_storage(stored)
    if not src.is_file():
        raise RuntimeError("Source video file is missing on disk")

    _upsert(video_id, status="processing", error=None)
    try:
        result = analyze_video(src, duration, ctx.progress, language=language, use_gemini=use_gemini)
    except JobCancelled:
        _upsert(video_id, status="cancelled")
        raise
    except Exception as e:
        _upsert(video_id, status="failed", error=(str(e) or e.__class__.__name__)[:500])
        raise

    _upsert(
        video_id,
        status="ready",
        language=result["language"],
        transcript_json=json.dumps(result["segments"], ensure_ascii=False),
        scenes_json=json.dumps(result["scenes"]),
        highlights_json=json.dumps(result["highlights"], ensure_ascii=False),
        scoring_source=result["scoring_source"],
        note=result["note"] or None,
        error=None,
    )
