def folder_size(path: Path) -> int:
    total = 0
    for dirpath, _dirs, files in os.walk(path):
        for f in files:
            try:
                total += os.path.getsize(os.path.join(dirpath, f))
            except OSError:
                pass
    return total

@app.get("/api/dashboard")
def dashboard(db: Session = Depends(get_db)):
    def count(model, *conds):
        stmt = select(func.count()).select_from(model)
        if conds:
            stmt = stmt.where(*conds)
        return db.scalar(stmt) or 0

    recent_videos = db.scalars(select(Video).order_by(Video.created_at.desc()).limit(5)).all()
    recent_clips = db.scalars(select(Clip).order_by(Clip.created_at.desc()).limit(5)).all()
    return {
        "total_videos": count(Video),
        "total_clips": count(Clip),
        "completed_jobs": count(Job, Job.status == "completed"),
        "processing_jobs": count(Job, Job.status == "processing"),
        "failed_jobs": count(Job, Job.status == "failed"),
        "storage_bytes": folder_size(STORAGE_DIR),
        "recent_videos": [
            {"id": v.id, "name": v.original_name, "created_at": v.created_at.isoformat()}
            for v in recent_videos
        ],
        "recent_clips": [
            {"id": c.id, "title": c.title, "created_at": c.created_at.isoformat()}
            for c in recent_clips
        ],
    }

def ensure_storage_dirs() -> None:
    for d in SUBDIRS:

def safe_display_name(name: str) -> str:
    """Only used for DISPLAY. Files on disk always get random UUID names."""
    name = os.path.basename(name.replace("\\", "/"))
    name = re.sub(r'[\x00-\x1f<>:"|?*]', "_", name).strip(" .")
    return name[:200] or "video"

import shutil
import uuid
from pathlib import Path

from fastapi import APIRouter, Depends, File, HTTPException, UploadFile
from fastapi.responses import FileResponse, JSONResponse
from sqlalchemy import func, select
from sqlalchemy.orm import Session

from backend import config
from backend.database.db import get_db
from backend.database.models import Video
from backend.video_processing.ffprobe import ProbeError, probe

router = APIRouter(prefix="/api/videos", tags=["videos"])
CHUNK = 1024 * 1024


class UploadRejected(Exception):
    def __init__(self, code: str, message: str):
        self.code = code
        self.message = message


def video_dict(v: Video) -> dict:
    return {
        "id": v.id,
        "name": v.original_name,
        "size_bytes": v.size_bytes,
        "duration_sec": v.duration_sec,
        "width": v.width,
        "height": v.height,
        "format": v.format,
        "status": v.status,
        "has_video": v.width is not None,
        "created_at": v.created_at.isoformat(),
    }


def _save_one(upload: UploadFile, db: Session) -> dict:
    display = config.safe_display_name(upload.filename or "")
    ext = Path(display).suffix.lower()
    if ext not in config.ALLOWED_EXTENSIONS:
        allowed = ", ".join(sorted(config.ALLOWED_EXTENSIONS))
        raise UploadRejected("bad_type", f"Unsupported file type '{ext or 'none'}'. Allowed: {allowed}")

    tmp = config.STORAGE_DIR / "temporary" / f"{uuid.uuid4().hex}.part"
    limit = config.max_upload_bytes()
    size = 0
    try:
        with open(tmp, "wb") as out:
            while True:
                chunk = upload.file.read(CHUNK)
                if not chunk:
                    break
                size += len(chunk)
                if size > limit:
                    raise UploadRejected("too_large", "File is larger than the allowed upload limit (MAX_UPLOAD_GB in .env)")
                out.write(chunk)
        if size == 0:
            raise UploadRejected("empty", "File is empty")
        try:
            info = probe(tmp)
        except ProbeError as e:
            raise UploadRejected("invalid_media", str(e))

        final_name = f"{uuid.uuid4().hex}{ext}"
        final = config.STORAGE_DIR / "original_videos" / final_name
        shutil.move(str(tmp), str(final))
    finally:
        tmp.unlink(missing_ok=True)

    try:
        v = Video(
            original_name=display,
            stored_path=f"original_videos/{final_name}",
            size_bytes=size,
            duration_sec=info["duration_sec"],
            width=info["width"],
            height=info["height"],
            format=ext.lstrip("."),
            status="uploaded",
        )
        db.add(v)
        db.commit()
        db.refresh(v)
    except Exception:
        db.rollback()
        final.unlink(missing_ok=True)
        raise
    return video_dict(v)


@router.post("/upload")
def upload_videos(files: list[UploadFile] = File(...), db: Session = Depends(get_db)):
    uploaded, errors = [], []
    for f in files:
        try:
            uploaded.append(_save_one(f, db))
        except UploadRejected as e:
            errors.append({"filename": f.filename, "code": e.code, "error": e.message})
    if uploaded or not errors:
        status = 200
    elif any(e["code"] == "too_large" for e in errors):
        status = 413
    else:
        status = 400
    return JSONResponse({"uploaded": uploaded, "errors": errors}, status_code=status)


@router.get("")
def list_videos(q: str = "", format: str = "", limit: int = 100, offset: int = 0, db: Session = Depends(get_db)):
    stmt = select(Video)
    if q.strip():
        stmt = stmt.where(Video.original_name.contains(q.strip(), autoescape=True))
    if format:
        stmt = stmt.where(Video.format == format.lower().lstrip("."))
    total = db.scalar(select(func.count()).select_from(stmt.subquery())) or 0
    rows = db.scalars(
        stmt.order_by(Video.created_at.desc(), Video.id.desc())
        .limit(min(max(limit, 1), 500))
        .offset(max(offset, 0))
    ).all()
    return {"total": total, "items": [video_dict(v) for v in rows]}


def _get_or_404(db: Session, video_id: int) -> Video:
    v = db.get(Video, video_id)
    if not v:
        raise HTTPException(404, "Video not found")
    return v


def _resolve(v: Video) -> Path:
    base = config.STORAGE_DIR.resolve()
    p = (base / v.stored_path).resolve()
    if base not in p.parents:
        raise HTTPException(400, "Invalid stored path")
    return p


@router.get("/{video_id}")
def get_video(video_id: int, db: Session = Depends(get_db)):
    return video_dict(_get_or_404(db, video_id))


@router.get("/{video_id}/stream")
def stream_video(video_id: int, db: Session = Depends(get_db)):
    v = _get_or_404(db, video_id)
    p = _resolve(v)
    if not p.is_file():
        raise HTTPException(404, "File is missing on disk")
    return FileResponse(p, media_type=config.MEDIA_TYPES.get(p.suffix.lower(), "application/octet-stream"))


@router.delete("/{video_id}")
def delete_video(video_id: int, db: Session = Depends(get_db)):
    v = _get_or_404(db, video_id)
    _resolve(v).unlink(missing_ok=True)
    db.delete(v)
    db.commit()
    return {"deleted": True, "id": video_id}

########## FILE: backend\video_processing\ffprobe.py ##########
import json
import shutil
import subprocess


class ProbeError(Exception):
    pass


def probe(path) -> dict:
    exe = shutil.which("ffprobe")
    if not exe:
        raise ProbeError("ffprobe not found. Install FFmpeg and restart VS Code.")
    try:
        r = subprocess.run(
            [exe, "-v", "error", "-print_format", "json", "-show_format", "-show_streams", str(path)],
            capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=60,
        )
    except subprocess.TimeoutExpired:
        raise ProbeError("ffprobe timed out while reading the file")
    if r.returncode != 0:
        raise ProbeError("This is not a valid audio/video file")
    try:
        data = json.loads(r.stdout or "{}")
    except json.JSONDecodeError:
        raise ProbeError("Could not read media information")

    streams = data.get("streams", [])
    video = next(
        (s for s in streams
         if s.get("codec_type") == "video" and s.get("disposition", {}).get("attached_pic", 0) != 1),
        None,
    )
    audio = next((s for s in streams if s.get("codec_type") == "audio"), None)
    if not video and not audio:
        raise ProbeError("No audio or video stream found in this file")

    raw = data.get("format", {}).get("duration") or (video or audio).get("duration")
    try:
        duration = float(raw) if raw is not None else None
    except (TypeError, ValueError):
        duration = None

    return {
        "duration_sec": duration,
        "width": video.get("width") if video else None,
        "height": video.get("height") if video else None,
    }

def stored_files():
    d = STORAGE_DIR / "original_videos"
    return {p.name for p in d.iterdir() if p.is_file() and p.name != ".gitkeep"}


def no_temp_leftovers():
    return list((STORAGE_DIR / "temporary").glob("*.part")) == []

export const ALLOWED_EXT = [".mp4", ".mov", ".mkv", ".avi", ".webm", ".mp3", ".wav"];

export type VideoItem = {
  id: number;
  name: string;
  size_bytes: number;
  duration_sec: number | null;
  width: number | null;
  height: number | null;
  format: string | null;
  status: string;
  has_video: boolean;
  created_at: string;
};

export function fmtBytes(n: number): string {
  if (n < 1024) return `${n} B`;
  const units = ["KB", "MB", "GB", "TB"];
  let i = -1;
  do { n /= 1024; i++; } while (n >= 1024 && i < units.length - 1);
  return `${n.toFixed(1)} ${units[i]}`;
}

export function fmtDuration(sec: number | null): string {
  if (sec == null) return "-";
  const s = Math.round(sec);
  const h = Math.floor(s / 3600);
  const m = Math.floor((s % 3600) / 60);
  const r = s % 60;
  const pad = (n: number) => String(n).padStart(2, "0");
  return h > 0 ? `${h}:${pad(m)}:${pad(r)}` : `${m}:${pad(r)}`;
}

"use client";
import { useRef, useState } from "react";
import Link from "next/link";
import { UploadCloud, CheckCircle2, AlertTriangle, Loader2, Clock } from "lucide-react";
import { API, ALLOWED_EXT, fmtBytes, fmtDuration, type VideoItem } from "@/lib/api";

type Item = {
  key: string;
  file: File;
  status: "queued" | "uploading" | "done" | "error";
  progress: number;
  error?: string;
  video?: VideoItem;
};

export default function UploadPage() {
  const [items, setItems] = useState<Item[]>([]);
  const [dragging, setDragging] = useState(false);
  const chain = useRef<Promise<void>>(Promise.resolve());
  const inputRef = useRef<HTMLInputElement>(null);

  const patch = (key: string, p: Partial<Item>) =>
    setItems((prev) => prev.map((i) => (i.key === key ? { ...i, ...p } : i)));

  function uploadOne(item: Item): Promise<void> {
    return new Promise((resolve) => {
      const xhr = new XMLHttpRequest();
      xhr.open("POST", `${API}/api/videos/upload`);
      xhr.upload.onprogress = (e) => {
        if (e.lengthComputable) patch(item.key, { progress: Math.round((e.loaded / e.total) * 100) });
      };
      xhr.onload = () => {
        try {
          const data = JSON.parse(xhr.responseText);
          if (xhr.status >= 200 && xhr.status < 300 && data.uploaded?.length) {
            patch(item.key, { status: "done", progress: 100, video: data.uploaded[0] });
          } else {
            patch(item.key, {
              status: "error",
              error: data.errors?.[0]?.error ?? data.detail ?? `Upload failed (${xhr.status})`,
            });
          }
        } catch {
          patch(item.key, { status: "error", error: `Upload failed (${xhr.status})` });
        }
        resolve();
      };
      xhr.onerror = () => {
        patch(item.key, { status: "error", error: "Cannot reach the backend. Is it running?" });
        resolve();
      };
      const form = new FormData();
      form.append("files", item.file);
      patch(item.key, { status: "uploading", progress: 0 });
      xhr.send(form);
    });
  }

  function addFiles(list: FileList | File[]) {
    const fresh: Item[] = Array.from(list).map((file, idx) => {
      const ext = "." + (file.name.split(".").pop() ?? "").toLowerCase();
      const ok = ALLOWED_EXT.includes(ext);
      return {
        key: `${Date.now()}-${idx}-${file.name}`,
        file,
        status: ok ? "queued" : "error",
        progress: 0,
        error: ok ? undefined : `Unsupported file type "${ext}"`,
      } as Item;
    });
    setItems((prev) => [...fresh, ...prev]);
    for (const it of fresh) {
      if (it.status === "queued") chain.current = chain.current.then(() => uploadOne(it));
    }
  }

  return (
    <div className="max-w-3xl">
      <h1 className="text-2xl font-bold text-white">Upload Videos</h1>
      <p className="text-sm text-zinc-400">
        MP4, MOV, MKV, AVI, WebM, MP3, WAV. Files are saved permanently on this computer.
      </p>

      <div
        onDragOver={(e) => { e.preventDefault(); setDragging(true); }}
        onDragLeave={() => setDragging(false)}
        onDrop={(e) => { e.preventDefault(); setDragging(false); addFiles(e.dataTransfer.files); }}
        onClick={() => inputRef.current?.click()}
        className={`mt-6 cursor-pointer rounded-xl border-2 border-dashed p-12 text-center transition-colors ${
          dragging ? "border-violet-500 bg-violet-600/10" : "border-zinc-700 bg-zinc-900/40 hover:border-zinc-500"
        }`}
      >
        <UploadCloud className="mx-auto text-violet-400" size={42} />
        <p className="mt-3 font-medium text-white">Drag and drop videos here</p>
        <p className="text-sm text-zinc-400">or click to browse files (multiple allowed)</p>
        <input
          ref={inputRef}
          type="file"
          multiple
          accept={ALLOWED_EXT.join(",")}
          className="hidden"
          onChange={(e) => { if (e.target.files) addFiles(e.target.files); e.target.value = ""; }}
        />
      </div>

      <div className="mt-6 space-y-3">
        {items.map((it) => (
          <div key={it.key} className="rounded-lg border border-zinc-800 bg-zinc-900/60 p-4">
            <div className="flex items-center justify-between gap-3">
              <div className="min-w-0">
                <p className="truncate font-medium text-white">{it.file.name}</p>
                <p className="text-xs text-zinc-500">{fmtBytes(it.file.size)}</p>
              </div>
              <div className="shrink-0 text-sm">
                {it.status === "queued" && <span className="flex items-center gap-1 text-zinc-400"><Clock size={16} /> Waiting</span>}
                {it.status === "uploading" && (
                  <span className="flex items-center gap-1 text-violet-300">
                    <Loader2 size={16} className="animate-spin" />
                    {it.progress >= 100 ? "Validating with FFprobe..." : `${it.progress}%`}
                  </span>
                )}
                {it.status === "done" && <span className="flex items-center gap-1 text-emerald-400"><CheckCircle2 size={16} /> Saved</span>}
                {it.status === "error" && <span className="flex items-center gap-1 text-red-400"><AlertTriangle size={16} /> Failed</span>}
              </div>
            </div>

            {it.status === "uploading" && (
              <div className="mt-3 h-1.5 overflow-hidden rounded bg-zinc-800">
                <div className="h-full bg-violet-500 transition-all" style={{ width: `${it.progress}%` }} />
              </div>
            )}
            {it.status === "done" && it.video && (
              <p className="mt-2 text-xs text-zinc-400">
                Duration {fmtDuration(it.video.duration_sec)}
                {it.video.width ? ` | ${it.video.width}x${it.video.height}` : " | audio only"}
              </p>
            )}
            {it.status === "error" && <p className="mt-2 text-xs text-red-400">{it.error}</p>}
          </div>
        ))}
      </div>

      {items.some((i) => i.status === "done") && (
        <Link href="/library" className="mt-6 inline-block rounded-lg bg-violet-600 px-4 py-2 text-sm font-medium text-white hover:bg-violet-500">
          Open Video Library
        </Link>
      )}
    </div>
  );
}

"use client";
import { useCallback, useEffect, useState } from "react";
import { Search, Trash2, Play, X, Film, Music } from "lucide-react";
import { API, fmtBytes, fmtDuration, type VideoItem } from "@/lib/api";

export default function LibraryPage() {
  const [items, setItems] = useState<VideoItem[]>([]);
  const [total, setTotal] = useState(0);
  const [q, setQ] = useState("");
  const [error, setError] = useState(false);
  const [loading, setLoading] = useState(true);
  const [preview, setPreview] = useState<VideoItem | null>(null);

  const load = useCallback(async (query: string) => {
    try {
      const r = await fetch(`${API}/api/videos?q=${encodeURIComponent(query)}`);
      if (!r.ok) throw new Error();
      const d = await r.json();
      setItems(d.items);
      setTotal(d.total);
      setError(false);
    } catch {
      setError(true);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    const t = setTimeout(() => load(q), 250);
    return () => clearTimeout(t);
  }, [q, load]);

  async function remove(v: VideoItem) {
    if (!window.confirm(`Permanently delete "${v.name}" from your computer?\nThis cannot be undone.`)) return;
    const r = await fetch(`${API}/api/videos/${v.id}`, { method: "DELETE" });
    if (r.ok) {
      if (preview?.id === v.id) setPreview(null);
      load(q);
    }
  }

  return (
    <div>
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-bold text-white">Video Library</h1>
          <p className="text-sm text-zinc-400">{total} file{total === 1 ? "" : "s"} saved locally</p>
        </div>
        <div className="relative">
          <Search size={16} className="absolute left-3 top-2.5 text-zinc-500" />
          <input
            value={q}
            onChange={(e) => setQ(e.target.value)}
            placeholder="Search by name..."
            className="w-64 rounded-lg border border-zinc-700 bg-zinc-900 py-2 pl-9 pr-3 text-sm text-white placeholder-zinc-500 outline-none focus:border-violet-500"
          />
        </div>
      </div>

      {error && (
        <div className="mt-6 rounded-lg border border-red-900 bg-red-950/40 p-4 text-sm text-red-300">
          Backend is offline. Start it with <code>.\scripts\start-all.ps1</code>
        </div>
      )}

      <div className="mt-6 overflow-hidden rounded-xl border border-zinc-800 bg-zinc-900/60">
        <table className="w-full text-left text-sm">
          <thead className="border-b border-zinc-800 text-xs uppercase tracking-wider text-zinc-500">
            <tr>
              <th className="px-4 py-3">Name</th>
              <th className="px-4 py-3">Duration</th>
              <th className="px-4 py-3">Resolution</th>
              <th className="px-4 py-3">Size</th>
              <th className="px-4 py-3">Uploaded</th>
              <th className="px-4 py-3 text-right">Actions</th>
            </tr>
          </thead>
          <tbody>
            {items.map((v) => (
              <tr key={v.id} className="border-b border-zinc-800/60 last:border-0 hover:bg-zinc-800/40">
                <td className="px-4 py-3">
                  <div className="flex items-center gap-2 text-white">
                    {v.has_video ? <Film size={16} className="text-violet-400" /> : <Music size={16} className="text-sky-400" />}
                    <span className="max-w-xs truncate">{v.name}</span>
                  </div>
                </td>
                <td className="px-4 py-3 text-zinc-300">{fmtDuration(v.duration_sec)}</td>
                <td className="px-4 py-3 text-zinc-300">{v.width ? `${v.width}x${v.height}` : "audio"}</td>
                <td className="px-4 py-3 text-zinc-300">{fmtBytes(v.size_bytes)}</td>
                <td className="px-4 py-3 text-zinc-400">{new Date(v.created_at).toLocaleString()}</td>
                <td className="px-4 py-3">
                  <div className="flex justify-end gap-2">
                    <button onClick={() => setPreview(v)} className="rounded-md bg-zinc-800 p-2 hover:bg-zinc-700" title="Preview">
                      <Play size={15} />
                    </button>
                    <button onClick={() => remove(v)} className="rounded-md bg-zinc-800 p-2 text-red-400 hover:bg-red-950" title="Delete">
                      <Trash2 size={15} />
                    </button>
                  </div>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
        {!loading && items.length === 0 && !error && (
          <p className="p-8 text-center text-sm text-zinc-500">
            {q ? "No files match your search." : "No files yet. Go to Upload Videos to add some."}
          </p>
        )}
      </div>

      {preview && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/80 p-6" onClick={() => setPreview(null)}>
          <div className="w-full max-w-3xl rounded-xl border border-zinc-700 bg-zinc-900 p-4" onClick={(e) => e.stopPropagation()}>
            <div className="mb-3 flex items-center justify-between">
              <p className="truncate font-medium text-white">{preview.name}</p>
              <button onClick={() => setPreview(null)} className="text-zinc-400 hover:text-white"><X size={20} /></button>
            </div>
            {preview.has_video ? (
              <video key={preview.id} controls autoPlay className="max-h-[60vh] w-full rounded bg-black" src={`${API}/api/videos/${preview.id}/stream`} />
            ) : (
              <audio key={preview.id} controls autoPlay className="w-full" src={`${API}/api/videos/${preview.id}/stream`} />
            )}
            <p className="mt-3 text-xs text-zinc-500">
              If a MOV/MKV/AVI file does not play here, it is still saved correctly. Browser preview for those formats improves in Phase 3.
            </p>
          </div>
        </div>
      )}
    </div>
  );
}
